let reviewFilter = 'all';

   ========================================================= */
const $ = id => document.getElementById(id);

const resultScreen= $('resultScreen');

/* =========================================================
   TIME → QUESTION COUNT MAPPING
   ========================================================= */
function getQuestionCount(minutes){
  if(minutes === 30) return 60;
  if(minutes === 60) return 90;
  if(minutes === 90) return 120;
  if(minutes === 120) return 150;
  return QUIZ_BANK.length; // 0 = no timer → all 200
}

/* =========================================================
   TIME SELECT
   ========================================================= */
document.querySelectorAll('#timeSelect button').forEach(btn=>{
  btn.addEventListener('click',()=>{
    document.querySelectorAll('#timeSelect button').forEach(b=>b.classList.remove('active'));
    btn.classList.add('active');
    timeLimit = parseInt(btn.dataset.min) * 60;
  });
});

/* =========================================================
   START / RESTART
   ========================================================= */
$('startBtn').addEventListener('click', startQuiz);
$('retryTopBtn').addEventListener('click', ()=>{
  if(confirm('क्या आप इसी क्विज़ को दोबारा शुरू करना चाहते हैं? आपके सभी उत्तर मिट जाएँगे।')){
    startQuiz();
  }
});

function startQuiz(){
  // Determine number of questions based on time limit
  const minutes = timeLimit / 60;
  const count = Math.min(getQuestionCount(minutes), QUIZ_BANK.length);

  // Shuffle a copy of the bank and pick 'count' questions
  const shuffled = [...QUIZ_BANK].sort(()=>Math.random()-0.5);
  activeQuiz = shuffled.slice(0, count);

  answers = new Array(activeQuiz.length).fill(null);
  hinted  = new Array(activeQuiz.length).fill(false);
  current = 0;
  submitted = false;
  reviewFilter = 'all';
  timeLeft = timeLimit;
  elapsed = 0;
  startTime = Date.now();

  // Reset UI
  startScreen.style.display = 'none';
  resultScreen.style.display = 'none';
  topBar.style.display = 'block';
  quizArea.style.display = 'block';
  $('palette').classList.remove('show');
  $('paletteToggle').textContent = '📋 सभी प्रश्न देखें (प्रश्न पैलेट)';

  $('totalNum').textContent = activeQuiz.length;
  buildPalette();
  renderQuestion();
  updateProgress();

  if(timerId) clearInterval(timerId);
  if(timeLimit > 0){
    $('timer').style.display = 'block';
    updateTimerDisplay();
    timerId = setInterval(()=>{
      timeLeft--;
      updateTimerDisplay();
      if(timeLeft <= 0){
        clearInterval(timerId);
        timerId = null;
        showModal('समय समाप्त! ⏰','निर्धारित समय पूरा हो गया है। आपके अब तक दिए गए उत्तरों के आधार पर परिणाम दिखाया जाएगा।',()=>{
          hideModal();
          finishQuiz(true);
        },true);
      }
    },1000);
  } else {
    $('timer').style.display = 'none';
    timerId = setInterval(()=>{ elapsed++; },1000);
  }
  window.scrollTo({top:0,behavior:'smooth'});
}

function updateTimerDisplay(){
  const el = $('timer');
  const m = Math.floor(timeLeft/60);
  const s = timeLeft % 60;
  el.textContent = `${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
  el.classList.remove('warn','danger');
  if(timeLeft <= 60) el.classList.add('danger');
  else if(timeLeft <= 300) el.classList.add('warn');
}

/* =========================================================
   RENDER QUESTION
   ========================================================= */
function renderQuestion(){
  const q = activeQuiz[current];
  $('curNum').textContent = current + 1;

  $('qText').innerHTML = `<span class="qnum">प्रश्न ${current+1}</span>${escapeHtml(q.q)}`;

  const opts = $('options');
  opts.innerHTML = '';
  const chosen = answers[current];

  q.o.forEach((text,i)=>{
    const div = document.createElement('div');
    div.className = 'opt';
    if(chosen !== null){
      div.classList.add('locked');
      if(i === q.a) div.classList.add('correct');
      else if(i === chosen) div.classList.add('wrong');
    }
    div.innerHTML = `<div class="key">${LETTERS[i]}</div>
                     <div style="flex:1">${escapeHtml(text)}</div>
                     ${chosen!==null && i===q.a ? '<div class="tick">✔</div>' : ''}
                     ${chosen!==null && i===chosen && i!==q.a ? '<div class="tick">✘</div>' : ''}`;
    if(chosen === null){
      div.addEventListener('click',()=>selectOption(i));
    }
    opts.appendChild(div);
  });

  const hb = $('hintBox');
  if(hinted[current]){
    hb.textContent = '💡 ' + q.h;
    hb.classList.add('show');
    $('hintBtn').textContent = '💡 हिंट छिपाएँ';
  } else {
    hb.classList.remove('show');
    $('hintBtn').textContent = '💡 हिंट देखें';
  }

  $('prevBtn').disabled = (current === 0);
  $('nextBtn').textContent = (current === activeQuiz.length - 1) ? 'अंतिम प्रश्न' : 'अगला →';

  updatePalette();
  updateProgress();
}

function escapeHtml(t){
  return String(t).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

/* =========================================================
   SELECT OPTION
   ========================================================= */
function selectOption(i){
  if(submitted) return;
  if(answers[current] !== null) return;
  answers[current] = i;
  renderQuestion();
}

/* =========================================================
   HINT
   ========================================================= */
$('hintBtn').addEventListener('click',()=>{
  hinted[current] = !hinted[current];
  const hb = $('hintBox');
  if(hinted[current]){
    hb.textContent = '💡 ' + activeQuiz[current].h;
    hb.classList.add('show');
    $('hintBtn').textContent = '💡 हिंट छिपाएँ';
  } else {
    hb.classList.remove('show');
    $('hintBtn').textContent = '💡 हिंट देखें';
  }
});

/* =========================================================
   NAVIGATION
   ========================================================= */
$('prevBtn').addEventListener('click',()=>{
  if(current > 0){ current--; renderQuestion(); scrollTopCard(); }
});
$('nextBtn').addEventListener('click',()=>{
  if(current < activeQuiz.length - 1){ current++; renderQuestion(); scrollTopCard(); }
  else { askSubmit(); }
});
function scrollTopCard(){ window.scrollTo({top:0,behavior:'smooth'}); }

/* =========================================================
   PALETTE
   ========================================================= */
function buildPalette(){
  const p = $('palette');
  p.innerHTML = '';
  for(let i=0;i<activeQuiz.length;i++){
    const b = document.createElement('button');
    b.textContent = i+1;
    b.dataset.i = i;
    b.addEventListener('click',()=>{
      current = i;
      renderQuestion();
      scrollTopCard();
    });
    p.appendChild(b);
  }
}
function updatePalette(){
  const p = $('palette');
  [...p.children].forEach((b,i)=>{
    b.classList.remove('correct','wrong','current');
    if(answers[i] !== null){
      b.classList.add(answers[i] === activeQuiz[i].a ? 'correct' : 'wrong');
    }
    if(i === current) b.classList.add('current');
  });
}
$('paletteToggle').addEventListener('click',()=>{
  const p = $('palette');
  p.classList.toggle('show');
  $('paletteToggle').textContent = p.classList.contains('show')
    ? '📋 प्रश्न पैलेट छिपाएँ'
    : '📋 सभी प्रश्न देखें (प्रश्न पैलेट)';
});

/* =========================================================
   PROGRESS
   ========================================================= */
function updateProgress(){
  const attempted = answers.filter(a=>a!==null).length;
  const pct = Math.round((attempted / activeQuiz.length) * 100);
  $('progressFill').style.width = pct + '%';
  $('answeredInfo').textContent = `उत्तर दिए: ${attempted} / ${activeQuiz.length}`;
  $('percentInfo').textContent = pct + '%';
}

/* =========================================================
   SUBMIT
   ========================================================= */
$('submitTop').addEventListener('click', askSubmit);

function askSubmit(){
  const attempted = answers.filter(a=>a!==null).length;
  const left = activeQuiz.length - attempted;
  let msg;
  if(left === 0){
    msg = `आपने सभी ${activeQuiz.length} प्रश्नों के उत्तर दे दिए हैं। क्या आप परिणाम देखना चाहते हैं?`;
  } else if(attempted === 0){
    msg = 'आपने अभी कोई उत्तर नहीं दिया है। सबमिट करने पर 0 अंक दिखेंगे। क्या आप सबमिट करना चाहते हैं?';
  } else {
    msg = `आपने ${attempted} प्रश्नों का उत्तर दिया है और ${left} प्रश्न छूटे हुए हैं। सबमिट करने पर केवल दिए गए उत्तरों का ही स्टेटस दिखाया जाएगा। (सही +2, गलत –1)`;
  }
  showModal('सबमिट करें?', msg, ()=>{ hideModal(); finishQuiz(false); });
}

function finishQuiz(timeUp){
  submitted = true;
  if(timerId){ clearInterval(timerId); timerId = null; }
  if(timeLimit === 0){ elapsed = Math.floor((Date.now() - startTime)/1000); }
  else { elapsed = timeLimit - timeLeft; }

  let correct = 0, wrong = 0, skipped = 0;
  answers.forEach((a,i)=>{
    if(a === null) skipped++;
    else if(a === activeQuiz[i].a) correct++;
    else wrong++;
  });
  const attempted = correct + wrong;
  const score = (correct * 2) - (wrong * 1);
  const maxScore = activeQuiz.length * 2;
  const pct = maxScore > 0 ? Math.round((score / maxScore) * 100) : 0;

  $('stTotal').textContent = activeQuiz.length;
  $('stAttempted').textContent = attempted;
  $('stCorrect').textContent = correct;
  $('stWrong').textContent = wrong;
  $('stSkipped').textContent = skipped;
  $('stTime').textContent = fmtTime(elapsed);

  $('pctText').textContent = pct + '%';
  $('fractionText').textContent = `${correct} / ${activeQuiz.length} सही`;
  $('scoreNote').textContent = `स्कोर: ${score} / ${maxScore}  (सही ×2, गलत ×–1)`;

  let title, msg;
  if(pct >= 85){ title='🏆 शानदार प्रदर्शन!'; msg='आपकी तैयारी बहुत मजबूत है। बधाई हो!'; }
  else if(pct >= 70){ title='🎉 बहुत अच्छा!'; msg='आपने अच्छा प्रदर्शन किया। कुछ विषयों पर और ध्यान दें।'; }
  else if(pct >= 50){ title='👍 ठीक-ठाक!'; msg='आधे से ज़्यादा सही हैं। रिवीज़न से सुधार संभव है।'; }
  else if(pct >= 30){ title='📚 अभ्यास की ज़रूरत है'; msg='आधार अच्छा है, लेकिन और अभ्यास करें।'; }
  else { title='🔁 दोबारा प्रयास करें'; msg='चिंता न करें — क्विज़ दोबारा लेकर सुधार करें।'; }

  if(timeUp) msg = 'समय समाप्त हो गया। ' + msg;

  $('resultTitle').textContent = title;
  $('statusMsg').textContent = msg;

  renderReview();

  topBar.style.display = 'none';
  quizArea.style.display = 'none';
  resultScreen.style.display = 'block';
  window.scrollTo({top:0,behavior:'smooth'});
}

function fmtTime(sec){
  const m = Math.floor(sec/60), s = sec % 60;
  return `${m}:${String(s).padStart(2,'0')}`;
}

function renderReview(){
  const list = $('reviewList');
  list.innerHTML = '';
  let shown = 0;

  activeQuiz.forEach((q,i)=>{
    const a = answers[i];
    let type;
    if(a === null) type = 'skip';
    else if(a === q.a) type = 'correct';
    else type = 'wrong';

    if(reviewFilter !== 'all' && reviewFilter !== type) return;
    shown++;

    const div = document.createElement('div');
    div.className = 'reviewItem ' + (type==='correct'?'ok':type==='wrong'?'bad':'skip');

    const your = a === null ? '— उत्तर नहीं दिया —' : `${LETTERS[a]}) ${q.o[a]}`;
    const right = `${LETTERS[q.a]}) ${q.o[q.a]}`;

    let badge = type==='correct' ? '<span style="color:#00e08c;font-weight:800;">✔ सही (+2)</span>'
              : type==='wrong'   ? '<span style="color:#ff5a5a;font-weight:800;">✘ गलत (–1)</span>'
              : '<span style="color:#ffd86f;font-weight:800;">○ छोड़ा (0)</span>';

    div.innerHTML = `
      <div class="rq">${i+1}. ${escapeHtml(q.q)} ${badge}</div>
      <div class="ra">आपका उत्तर: <span class="${type==='correct'?'good':type==='wrong'?'bad':''}">${escapeHtml(your)}</span></div>
      ${type!=='correct' ? `<div class="ra">सही उत्तर: <span class="good">${escapeHtml(right)}</span></div>` : ''}
      <div class="ra" style="margin-top:4px;color:#8fa4b8;">💡 ${escapeHtml(q.h)}</div>
    `;
    list.appendChild(div);
  });

  if(shown === 0){
    list.innerHTML = '<p style="color:#9fb3c8;font-size:14px;text-align:center;padding:18px;">इस श्रेणी में कोई प्रश्न नहीं है।</p>';
  }
}
